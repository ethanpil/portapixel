# PortaPixel Raspberry Pi firmware settings. This file is the only source
# (plan section 6). build-image.sh copies it to PPBOOT with no change.
#
# One aarch64 image covers Pi Zero 2 W, Pi 2 v1.2, Pi 3, Pi 4, Pi 5 and the
# Compute Modules. The Pi firmware picks the right device tree for the board, so
# no per-model file is needed (D8).

[all]
# Alpine names the kernel and the initramfs after the flavour.
kernel=vmlinuz-rpi
initramfs initramfs-rpi followkernel

# 64 bit only. There is no armv7 or ARMv6 image (D8).
arm_64bit=1

# The 3.5 mm audio jack, for [audio].output = "analog" (D11). HDMI audio needs
# nothing here: it comes through the vc4 driver.
# This line MUST stay above the first dtoverlay line. A dtparam line below a
# dtoverlay line goes to that overlay, and vc4-kms-v3d has an "audio"
# parameter of its own (HDMI audio). There, the line never reached the jack.
dtparam=audio=on

# Use the whole panel. A TV that adds a black border makes signage look broken.
disable_overscan=1

# Serial console on the GPIO header. It is the last way into a box with no
# network (plan section 5). The kernel resolves "serial0" in cmdline.txt through
# the device tree, so the same line works on every Pi model.
enable_uart=1

# No rainbow splash and no boot delay: the screen must show content, not our
# boot (D18).
disable_splash=1
boot_delay=0

# The KMS driver for the VideoCore GPU. mpv draws directly on DRM/KMS (D3), so
# this overlay is what makes a picture possible at all. On a Pi 4 the firmware
# loads vc4-kms-v3d-pi4 for this name, and on a Pi 5 vc4-kms-v3d-pi5.
dtoverlay=vc4-kms-v3d
# Two framebuffers, which KMS needs for a page flip.
max_framebuffers=2

# CMA, the contiguous memory of the kernel. The display buffers and the video
# decoder (bcm2835-codec) take their buffers from it. The defaults of the
# overlay (read from the dtbo files of linux-rpi 6.12.85): vc4-kms-v3d 256 MB,
# vc4-kms-v3d-pi4 508 MB, vc4-kms-v3d-pi5 64 MB (a Pi 5 has an IOMMU).
#
# 256 MB is half the memory of a 512 MB board, and the overlay documentation
# says that 192 MB and more need 1 GB. So a 512 MB board gets 128 MB, the
# largest size that the documentation permits there. The estimate for 1080p:
# about 25 MB for three scanout buffers of 8.3 MB, about 8 MB for the console
# framebuffer, about 75 MB for the decoder (20 frames of 3.1 MB, which is the
# FFmpeg default, and the bitstream buffers), and the GL textures of mpv
# --vo=gpu (the GPU of a Pi 0 to Pi 3 takes all its memory from CMA). A free
# CMA page is not lost: the kernel puts movable pages, also zram pages, there.
# NOT PROVEN on a real Pi.
#
# A dtparam line here goes to the overlay above (see the audio line). [pi02]
# is the Zero 2 W and the Compute Module 0. board-type 0x0e is the Pi 3 A+.
# Every other board keeps the default of its overlay.
[pi02]
dtparam=cma-128
[board-type=0x0e]
dtparam=cma-128
[all]

# NOT set, on purpose:
#   hdmi_force_hotplug  -- it makes a fake connector. "No display" must stay a
#                          real wait state that recovers on hotplug (D44).
#   hdmi_group/hdmi_mode -- the EDID is trusted. A display that lies is fixed
#                          with the [display].video_mode setting (D49).
#   gpu_mem             -- the KMS driver takes what it needs. A fixed split
#                          only wastes RAM on a 512 MB Pi Zero 2 W. The H.264
#                          decoder of the firmware must work in the default
#                          split (64 MB, 76 MB on a Pi 4). NOT PROVEN on a
#                          real Pi.
